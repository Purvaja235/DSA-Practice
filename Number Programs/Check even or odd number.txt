CHECK EVEN AND ODD NUMBER

1.A => Using Modulus Operator (%)
           int a = 15;
	       int temp = a;
	       
	       while(temp > 1){
	           temp = temp - 2;
	       }
	       if(temp == 0){
	           System.out.println("Even");
	       }
	       else {
	           System.out.println("Odd");
	       }
		Time Complexity=> O(n) 
		Space Complexity => O(1)
		
2.A =>	Using Bitwise Operator (&)
          if((a &1) == 0){
	           System.out.println("Even");
	       }
	       else {
	           System.out.println("Odd");
	       }
		Time Complexity => O(1)
		Space Complexity => O(1)

3.A =>  Using Ternary Operator
          String s = (a % 2 == 0) ? "Even" : "Odd";
	        System.out.println(s);
        Time Complexity => O(1)
        Space Complexity => O(1)
