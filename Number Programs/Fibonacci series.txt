Fibonacci series

1.A=>  Approach 1: Iterative (Best for Freshers & Interviews)
           int n = 10;
	      
	      int a = 0;
	      int b = 1;
	      System.out.print(a + " " + b + " ");
	      for(int i = 3; i<= n; i++){
	          int c = a + b;
	           System.out.print(c + " ");
	          a = b;
	          b = c;
	         
	      }
		Time Complexity => O(n)
        Space Complexity => O(1)

2.A=>  	Using Array
           int n = 10;
	     int[] fib = new int[n];
	     fib[0] = 0;
	     fib[1] = 1;
	     for(int i = 2; i < n ; i++){
	         fib[i] = fib[i - 1] + fib[i - 2];
	     }
	     for(int num : fib){
	         System.out.print(num + " ");
	     }	
		 Time Complexity => O(n)
         Space Complexity => O(n)

3.A=>   Recursion (Interview Follow-Up)
         static int fib(int n){
	         if(n <= 1) {
	             return n;
	         }
	         return fib(n - 1) + fib(n - 2);
	     }
	     public static void main(String args[]){
	         int n = 10;
	         for(int i = 0; i < n; i++){
	             System.out.println(fib(i) + " ");
	         }
	     }		 
		 Time Complexity => O(2^n)
         Space Complexity => O(n)
