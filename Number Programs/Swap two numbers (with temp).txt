Swap two numbers (with temp)

1.A =>  Brute Force / Standard Interview Answer (Using Temp Variable)
        int a = 10;
	     int b = 13;
	     int temp = a;
	     a = b;
	     b = temp;
	     System.out.println("a = " + a);
        System.out.println("b = " + b);
		 
		Time Complexity => O(1)
        Space Complexity => O(1)

2.A =>  Without Temp (Addition/Subtraction)
        int a = 10;
	     int b = 13;
	     a = a + b;
	     b = a - b;
	     a = a - b;
	     System.out.println(a + " " + b);

        Time Complexity => O(1)
        Space Complexity => O(1)

3.A =>  Without Temp (Multiplication/Division)
        int a = 10;
	     int b = 13;
	     a = a * b;
	     b = a / b;
	     a = a / b;
	     System.out.println(a + " " + b);        		 
		 
		Time Complexity => O(1)
        Space Complexity => O(1)
