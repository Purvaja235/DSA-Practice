Check strong number

1.A =>  Brute Force / Standard Approach
          int n = 145;
	    int sum = 0;
	    int temp = n;
	    while(temp > 0){
	       int digit = temp % 10;
	       int factorial = 1;
	       for(int i = 1; i <= digit; i++){
	           factorial = factorial * i;
	       }
	       sum += factorial; 
	        
	        temp = temp / 10;
	     }
	    if(n == sum){
	      System.out.println("Strong Number");
	    }
	    else {
	      System.out.println("Not");
	    }
	    Time Complexity => O(log n)
        Space Complexity => O(1)
	  
2.A =>  Most Optimized Approach (Precompute Factorials)
        int[] fact = {1,1,2,6,24,120,720,5040,40320,362880};

        int n = 145;
        int original = n;
        int sum = 0;

        while (n > 0) {
            int digit = n % 10;
            sum += fact[digit];
            n /= 10;
        }

        if (sum == original) {
            System.out.println("Strong Number");
        } else {
            System.out.println("Not Strong Number");
        }	  
		Time Complexity => O(log n)
        Space Complexity => O(1)
