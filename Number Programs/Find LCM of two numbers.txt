Find LCM of two numbers

1.A =>  Brute Force
        Start from the bigger number.
        Check if it is divisible by both numbers.
        If not, increase by 1 and check again.
		
		int a = 18;
		int b = 12;
		int lcm = Math.max(a, b);
		while(true){
		    if(lcm % a == 0 && lcm % b == 0){
		        System.out.println(lcm);
		        break;
		    }
		    lcm++;
		}
		Time Complexity => O(LCM)
        Space Complexity => O(1)
		
2.A =>  Better
        Instead of increasing by 1, increase by the larger number
        int a = 18;
		int b = 12;
		int lcm = Math.max(a, b);
	    for(int i = lcm; ; i +=lcm) {
	        if(i % a == 0 && i % b == 0){
	            System.out.println(i);
	            break;
	        }
	    }
        Time Complexity = O(LCM / max(a,b))
        Space Complexity = O(1)	

3.A =>  Most Optimized (Using GCD)
        Formula = Find GCD then LCM = (a * b) / gcd
		int a = 18;
		int b = 12;
		int x1 = a;
		int x2 = b;
		while(x2 != 0){
		    int temp = x2;
		    x2 = x1 % x2;
		    x1 = temp;
		}
		System.out.println(x1);
		int lcm = a * b / x1;
		System.out.println(lcm);
		Time Complexity = O(log(min(a,b)))
        Space Complexity = O(1)
